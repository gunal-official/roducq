async function bodyOf(req) {
  const parts = [];
  for await (const chunk of req) parts.push(chunk);
  const raw = Buffer.concat(parts).toString("utf8");
  if (!raw) return {};
  try { return JSON.parse(raw); } catch { return { raw }; }
}
function send(req, res, data, status = 200, extra = {}) {
  res.writeHead(status, {
    "content-type": "application/json; charset=utf-8", "access-control-allow-origin": "*",
    "access-control-allow-methods": "GET,POST,PATCH,DELETE,OPTIONS",
    "access-control-allow-headers": req.headers["access-control-request-headers"] || "*", ...extra,
  });
  res.end(JSON.stringify(data));
}
function single(req, row) {
  // PostgREST's maybeSingle() accepts an array for zero/one records.
  if (req.headers.accept?.includes("application/vnd.pgrst.object+json")) return row ?? null;
  return row == null ? [] : [row];
}

async function listen(server) {
  await new Promise((resolve, reject) => {
    server.once("error", reject);
    server.listen(0, "127.0.0.1", resolve);
  });
  return server.address().port;
}
async function freePort() {
  const server = createNetServer();
  const port = await listen(server);
  await new Promise((resolve) => server.close(resolve));
  return port; // Next fails normally if another process claims it; we never terminate that process.
}
function run(command, args, env, logfile) {
  return new Promise((resolve, reject) => {
    const child = spawn(command, args, { cwd: ROOT, env, stdio: ["ignore", "pipe", "pipe"] });
    let log = "";
    child.stdout.on("data", (d) => { log += d; });
    child.stderr.on("data", (d) => { log += d; });
    child.once("error", reject);
    child.once("exit", (code, signal) => {
      writeFileSync(logfile, log);
      if (code === 0) resolve();
      else reject(new Error(`${command} exited ${code ?? signal}; see ${logfile}\n${log.slice(-5000)}`));
    });
  });
}

async function newContext(browser, base, stubOrigin, { width = 1440, scheme = "light", auth = true } = {}) {
  const context = await browser.newContext({ viewport: { width, height: HEIGHTS[width] }, colorScheme: scheme, deviceScaleFactor: 1,
    locale: "en-US", timezoneId: "UTC", reducedMotion: "reduce", serviceWorkers: "block" });
  if (auth) await context.addCookies([{ name: "sb-127-auth-token", value: sessionCookie(), url: base }]);
  await context.route("**/*", (route) => {
    const url = new URL(route.request().url());
    if (url.origin === base || url.origin === stubOrigin) return route.continue();
    return route.abort("blockedbyclient");
  });
  const page = await context.newPage();
  page.on("pageerror", (error) => browserErrors.push({ type: "pageerror", message: error.message }));
  page.on("response", (response) => {
    if (response.url().startsWith(base) && response.status() >= 400)
      browserErrors.push({ type: "http", status: response.status(), url: response.url() });
  });
  await installFixtureNetwork(page, base, stubOrigin);
  await context.grantPermissions(["local-network-access", "clipboard-read", "clipboard-write"], { origin: base });
  return { context, page };
}

function round(n) { return Math.round(n * 1000) / 1000; }
function luminance(rgb) {
  const c = rgb.map(v => v / 255).map(v => v <= .04045 ? v / 12.92 : ((v + .055) / 1.055) ** 2.4);
  return .2126 * c[0] + .7152 * c[1] + .0722 * c[2];
}
function ratio(a, b) {
  const [x, y] = [luminance(a), luminance(b)].sort((p,q) => q - p);
  return (x + .05) / (y + .05);
}
function parseRgb(s) {
  const m = s.match(/^rgba?\(\s*([\d.]+)[,\s]+([\d.]+)[,\s]+([\d.]+)/);
  assert.ok(m, `Unexpected foreground color: ${s}`);
  return m.slice(1, 4).map(Number);
}

    main.scrollTop=best?.scrollTop ?? 0;

      timerFitsViewport:t.top>=0&&t.bottom<=innerHeight&&t.left>=0&&t.right<=innerWidth};
  },phase);
}
async function timerProbe(page, base, width) {
  reset(); await goto(page,base,'/intake','light');
  await page.locator('#intake-context').waitFor({state:'visible'});
  const phases=[];
  for (const phase of ['idle','running','stop']) {
    if (phase==='running') await page.getByRole('button',{name:'Start timer'}).click();
    if (phase==='stop') await page.getByRole('button',{name:'Stop',exact:true}).click();
    const measure=await timerGeometry(page,phase);
    assert.ok(!measure.error, JSON.stringify(measure));
    await page.evaluate(() => new Promise(r=>requestAnimationFrame(()=>requestAnimationFrame(r))));
    const screenshot=`timer-${width}-${phase}.png`;
    await page.screenshot({path:path.join(OUT,screenshot),animations:'disabled'});
    phases.push({...measure,screenshot,activeFocus:await identifyFocus(page)});
  }
  const stopControls=await page.evaluate(()=>[...document.querySelectorAll('input,button,[role="combobox"]')]
    .filter(e=>e.closest('div') && (e.getAttribute('aria-label')==='What did you work on?'||e.getAttribute('aria-label')==='Minutes'||e.getAttribute('aria-label')==='Worked on'||e.getAttribute('aria-label')==='Brief'||['Save','Keep running'].includes(e.textContent.trim())))
    .map(e=>({tag:e.tagName,label:e.getAttribute('aria-label')||e.textContent.trim(),disabled:e.disabled,rect:(()=>{const r=e.getBoundingClientRect();return {x:+r.x.toFixed(2),y:+r.y.toFixed(2),width:+r.width.toFixed(2),height:+r.height.toFixed(2)};})()})));
  await page.getByRole('button',{name:'Keep running'}).click();
  assert.equal(await page.getByRole('button',{name:'Stop',exact:true}).count(),1);
  const storageWhileRunning=await page.evaluate(()=>Boolean(localStorage.getItem('roducq.timer.v1')));
  await page.getByRole('button',{name:'Discard this session'}).click();
  assert.equal(await page.getByRole('button',{name:'Start timer'}).count(),1);
  const storageAfterDiscard=await page.evaluate(()=>Boolean(localStorage.getItem('roducq.timer.v1')));
  return {width,height:HEIGHTS[width],scheme:'light',phases,stopControls,storageWhileRunning,storageAfterDiscard,fixtureWrites:state.writes};
}
